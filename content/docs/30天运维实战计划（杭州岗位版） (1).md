#### 脚本①：服务器日常巡检报告

```bash
#!/bin/bash
# daily_check.sh - 检查CPU/内存/磁盘/关键服务，输出巡检报告
REPORT_FILE="/var/log/ops/check_$(date +%Y%m%d_%H%M).txt"
mkdir -p /var/log/ops
WARN=0

report() { echo "$1" | tee -a $REPORT_FILE; }
warn()   { echo "[⚠️  警告] $1" | tee -a $REPORT_FILE; WARN=$((WARN+1)); }

report "===== 巡检报告 $(date) ====="

CPU=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)
report "CPU使用率: ${CPU}%"
[ $(echo "$CPU > 80" | bc) -eq 1 ] && warn "CPU超过80%"

MEM_PCT=$(free | awk 'NR==2{printf "%d", $3/$2*100}')
report "内存使用率: ${MEM_PCT}%"
[ $MEM_PCT -gt 85 ] && warn "内存超过85%"

df -h | awk 'NR>1' | while read line; do
    pct=$(echo $line | awk '{print $5}' | tr -d '%')
    mnt=$(echo $line | awk '{print $6}')
    report "磁盘 $mnt: ${pct}%"
    [ $pct -gt 80 ] && warn "磁盘 $mnt 超过80%"
done

for svc in nginx mysql redis docker; do
    systemctl is-active $svc > /dev/null 2>&1 \
        && report "服务 $svc: ✅ 运行中" \
        || warn "服务 $svc ❌ 未运行"
done

report "===== 共 $WARN 个警告 ====="
```

#### 脚本②：自动备份与清理

```bash
#!/bin/bash
# auto_backup.sh - 备份指定目录，保留7天，记录日志
BACKUP_SRC="/etc /var/log/nginx"
BACKUP_DEST="/data/backup"
DATE=$(date +%Y%m%d_%H%M%S)
LOG="/var/log/ops/backup.log"
mkdir -p $BACKUP_DEST

log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a $LOG; }

BACKUP_FILE="${BACKUP_DEST}/backup_${DATE}.tar.gz"
tar -czf $BACKUP_FILE $BACKUP_SRC 2>/dev/null && \
    log "备份成功: $BACKUP_FILE ($(du -sh $BACKUP_FILE | cut -f1))" || \
    { log "备份失败"; exit 1; }

DELETED=$(find $BACKUP_DEST -name "backup_*.tar.gz" -mtime +7 -delete -print | wc -l)
log "清理过期备份: ${DELETED}个"
```

#### 脚本③：批量检查多台服务器状态

```bash
#!/bin/bash
# batch_check.sh - SSH批量检查多台服务器资源状态
HOSTS=("192.168.10.10" "192.168.10.11" "192.168.10.12")
SSH_KEY="/root/.ssh/id_rsa"

printf "%-18s %-10s %-10s %-10s %-10s\n" "主机" "CPU%" "内存%" "磁盘%" "状态"
echo "--------------------------------------------------------------"
for HOST in "${HOSTS[@]}"; do
    RESULT=$(ssh -i $SSH_KEY -o ConnectTimeout=5 -o StrictHostKeyChecking=no root@${HOST} \
        "echo \$(top -bn1|grep Cpu|awk '{print \$2}'|cut -d%% -f1) \
         \$(free|awk 'NR==2{printf \"%d\", \$3/\$2*100}') \
         \$(df /|awk 'NR==2{print \$5}'|tr -d %%)" 2>/dev/null)
    if [ $? -eq 0 ]; then
        printf "%-18s %-10s %-10s %-10s %-10s\n" "$HOST" \
            "$(echo $RESULT|awk '{print $1}')%" \
            "$(echo $RESULT|awk '{print $2}')%" \
            "$(echo $RESULT|awk '{print $3}')%" "✅ 在线"
    else
        printf "%-18s %-10s %-10s %-10s %-10s\n" "$HOST" "-" "-" "-" "❌ 离线"
    fi
done
```

### 第5天：MySQL 主从复制 + 备份恢复

## 上午：主从搭建

### 1. 核心配置文件 (`my.cnf` / `my.ini`)

> **注意：** 必须确保两台服务器的 `server_id` 唯一。

```toml
# ================= 主库 (Source) 配置 =================
[mysqld]
server_id = 1
log_bin = mysql-bin
gtid_mode = ON
enforce_gtid_consistency = ON

# ================= 从库 (Replica) 配置 =================
[mysqld]
server_id = 2
gtid_mode = ON
enforce_gtid_consistency = ON
```

*修改后需重启两台 MySQL 服务。*

### 2. 主库 (Source) 端：创建并授权账号

> **注意：** MySQL 8.4 严禁使用 `GRANT` 隐式创建用户，必须分步执行。

```sql
-- 1. 创建复制专用用户（若密码策略报错，请保持这种高强度密码格式）
CREATE USER 'repl_user'@'%' IDENTIFIED BY 'Replica@2026_Secure';

-- 2. 独立授予复制权限
GRANT REPLICATION SLAVE ON *.* TO 'repl_user'@'%';

-- 3. 刷新权限
FLUSH PRIVILEGES;
```

### 3. 从库 (Replica) 端：一键绑定与启动

> **注意：** 8.4 版本已彻底废弃 `MASTER` / `SLAVE` 关键字，必须使用 `SOURCE` / `REPLICA`。如果重复配置，必须先执行 `STOP REPLICA;`。

```SQL
-- 1. 安全起见，先停止复制线程（防止报 3081 错误）
STOP REPLICA;

-- 2. 配置主库信息（基于 GTID 自动对齐，无需指定物理文件和 Pos 点）
CHANGE REPLICATION SOURCE TO
  SOURCE_HOST='192.168.11.129',
  SOURCE_USER='repl_user',
  SOURCE_PASSWORD='Replica@2026_Secure',
  SOURCE_AUTO_POSITION=1;

-- 3. 开启复制
START REPLICA;
```

### 4. 结果验证

执行以下命令：

```SQL
SHOW REPLICA STATUS\G;
```

**检查核心指标：**

- `Replica_IO_Running: Yes` —— 正常，正在从主库读取日志。
- `Replica_SQL_Running: Yes` —— 正常，正在本地回放日志。

### **5：验证数据同步**

在主库执行写入，立即在从库查询，能查到说明主从搭建成功：

```sql
-- 主库
USE ops_db;
INSERT INTO t_deploy (env, status) VALUES ('test-deploy', 1);

-- 从库（应立即能查到）
USE ops_db;
SELECT * FROM t_deploy ORDER BY id DESC LIMIT 1;
```

### 故意破坏练排错

每个场景都要**亲手触发再修复**，这是本节最重要的练习。

---

#### 破坏① server-id 与主库相同

**操作：** 将从库 `/etc/my.cnf` 中 `server-id` 改为 `1`，重启 MySQL，执行 `START REPLICA`。

**预期报错：**

```
Fatal error: The replica I/O thread stops because source and replica
have equal MySQL server ids; these ids must be different for
replication to work.
```

**修复步骤：**

```bash
# 1. 将 /etc/my.cnf 中 server-id 改回唯一值（如 2）
# 2. 重启 MySQL
systemctl restart mysqld
```

```sql
-- 3. 重新启动复制
START REPLICA;
SHOW REPLICA STATUS\G
```

---

#### 破坏② SOURCE_LOG_POS 填错

**操作：** 故意填一个不存在的 POS 值（如 `9999999`），观察 SQL 线程报错。

**预期现象：**

```sql
SHOW REPLICA STATUS\G
-- Replica_SQL_Running: No
-- Last_SQL_Error: 解析到无效事件或事务缺失，具体看 Last_SQL_Error 字段
```

**修复步骤：**

```sql
-- 停止复制并清除旧配置
STOP REPLICA;
RESET REPLICA;    -- 8.4 语法，旧的 RESET SLAVE 已移除

-- 填入正确的 binlog 位置（从主库 SHOW BINARY LOG STATUS 重新获取）
CHANGE REPLICATION SOURCE TO
  SOURCE_LOG_FILE = 'mysql-bin.000003',
  SOURCE_LOG_POS  = 1234;    -- 填入正确值

START REPLICA;
SHOW REPLICA STATUS\G
```

---

#### 破坏③ 从库直接写数据（主从数据不一致）

**操作：** 临时关闭 `read_only`，在从库直接写入数据。

> 注意：若开启了 `super_read_only = ON`，连 root 用户也无法写入，需先关闭。

```sql
-- 从库临时关闭只读（模拟误操作）
SET GLOBAL super_read_only = OFF;
SET GLOBAL read_only = OFF;

-- 直接在从库写入（与主库不一致的数据）
USE ops_db;
INSERT INTO t_deploy (name, status) VALUES ('rogue-write', 99);

-- 恢复只读
SET GLOBAL read_only = ON;
SET GLOBAL super_read_only = ON;
```

**检测不一致：**

```sql
-- 主库查
SELECT COUNT(*) FROM ops_db.t_deploy;

-- 从库查，对比差异
SELECT COUNT(*) FROM ops_db.t_deploy;
```

**修复方法（重新全量同步，最彻底）：**

```bash
# 1. 主库重新 dump ops_db
mysqldump -uroot -pRoot@123 \
    --single-transaction \
    --source-data=2 \
    ops_db > /backup/ops_db_fix.sql

# 2. 从文件头部找到主库 binlog 位置
head -50 /backup/ops_db_fix.sql | grep 'CHANGE'
```

```sql
-- 3. 从库停止复制，导入数据，重新建立复制关系
STOP REPLICA;

-- 导入数据（在 bash 中执行）
-- mysql -uroot -pRoot@123 ops_db < /backup/ops_db_fix.sql

  SOURCE_LOG_FILE = 'mysql-bin.000003',    -- 从 dump 文件头部获取

## 今日总结

- MySQL 8.4 将所有 `SLAVE` 关键字改为 `REPLICA`，`MASTER` 改为 `SOURCE`
- mysqldump 的 `--master-data` 参数改为 `--source-data`

# RDB 持久化（定时快照，适合冷备份）

save 300 10     # 300秒内有10次写操作则保存
dbfilename dump.rdb
dir /var/lib/redis

# AOF 持久化（记录每条写命令，数据更可靠）
appendonly yes
appendfsync everysec    # 每秒刷盘（性能和安全的平衡点）
appendfilename appendonly.aof
```

```groovy
pipeline {
    agent any
    parameters {
        choice(name: 'DEPLOY_ENV', choices: ['dev','test','staging','prod'], description: '部署环境')
        string(name: 'VERSION', defaultValue: 'latest', description: '镜像版本tag')
    }
    environment {
        IMAGE_NAME = "myapp"
        REGISTRY   = "registry.cn-hangzhou.aliyuncs.com/yourname"
    }
    stages {
        stage('检出代码') {
            steps { git branch: 'main', url: 'https://github.com/yourname/ops-practice.git' }
        }
        stage('构建镜像') {
            steps {
                script {
                    env.IMAGE_TAG = "${BUILD_NUMBER}"
                    sh "docker build -t ${IMAGE_NAME}:${env.IMAGE_TAG} ."
                }
            }
        }
        stage('推送镜像') {
            steps {
                withCredentials([usernamePassword(credentialsId: 'aliyun-registry',
                    usernameVariable: 'USER', passwordVariable: 'PASS')]) {
                    sh "docker login ${REGISTRY} -u $USER -p $PASS"
                    sh "docker push ${REGISTRY}/${IMAGE_NAME}:${env.IMAGE_TAG}"
                }
            }
        }
        stage('部署') {
            steps {
                sshPublisher(publishers: [sshPublisherDesc(
                    configName: "主机A",
                    transfers: [sshTransfer(execCommand:
                        "cd /opt/apps && ./deploy.sh ${params.DEPLOY_ENV} ${env.IMAGE_TAG}"
                    )]
                )])
            }
        }
        stage('健康验证') {
            steps {
                sh """
                    sleep 15
                    curl -sf http://192.168.10.10/health && echo "✅ 服务正常" || exit 1
                """
            }
        }
    }
    post {
        success { echo "✅ ${IMAGE_NAME}:${env.IMAGE_TAG} 已部署到 ${params.DEPLOY_ENV}" }
        failure  { echo "❌ 部署失败，请查看 Console Output" }
    }
}
```

#!/bin/bash
# 整体架构健康验证脚本（第24天写）
echo "===== 架构健康验证 $(date) ====="

check() {
    local name=$1; local cmd=$2
    eval "$cmd" > /dev/null 2>&1 \
        && echo "✅ $name: 正常" \
        || echo "❌ $name: 异常"
}

check "Nginx"       "curl -sf http://192.168.10.10/"
check "MySQL主库"   "mysqladmin -h 192.168.10.10 -uroot -pRoot@123 ping"

```
背景：独立搭建了覆盖开发/测试/预发/生产四套环境的完整运维平台

行动：
• MySQL主从（A主B从），mysqldump定时备份+阿里云OSS异地存储，配置主从延迟Prometheus告警

• Jenkins流水线实现：代码提交→构建Docker镜像→推阿里云Registry→部署→健康验证全自动
• Prometheus监控全栈（含MySQL Exporter / Redis Exporter），Grafana大屏，告警覆盖CPU/磁盘/主从延迟/内存
• ELK收集Nginx JSON格式日志，Kibana定位5xx错误和慢请求
• Shell巡检脚本（CPU/内存/磁盘/服务状态），cron每小时执行，异常写日志
• 阿里云ECS/OSS/云监控实操，AWS EC2/S3/CloudWatch实操

结果：具备独立维护中小型公司全套运维环境的能力，覆盖部署/监控/日志/中间件全链路
```

**高频面试问题：**

| 问题 | 答题要点 |
|------|---------|
| MySQL主从延迟怎么排查？ | `Seconds_Behind_Master` → binlog积压 → 从库机器性能 → 网络带宽 |
| 主从数据不一致怎么修复？ | STOP SLAVE → 校验工具 pt-table-checksum → 重新全量同步 |
| Redis AOF和RDB怎么选？ | AOF更可靠（最多丢1秒数据）/ RDB恢复更快 / 生产建议双开 |
| Redis内存满了怎么处理？ | `INFO memory`看使用率 → 检查`maxmemory-policy` → 清大Key → 扩容 |
| 多环境配置怎么管理差异？ | .env文件按环境隔离 + Docker Compose project + 密钥不入Git |
| Jenkins流水线失败怎么排查？ | Console Output定位失败Stage → 本地复现命令 → 检查凭证/权限/网络 |
| 零停机部署怎么实现？ | 多副本滚动更新 + 健康检查通过再切流量 + Nginx upstream摘流 |
| 阿里云和AWS主要区别？ | 产品命名/控制台风格/文档语言不同，核心概念一一对应，计费模式类似 |
